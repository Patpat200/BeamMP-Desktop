cmake_minimum_required(VERSION 3.21)
cmake_policy(SET CMP0091 NEW)
project(Launcher LANGUAGES CXX)
include(CTest)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(BEAMMP_BUILD_GUI "Build the optional Qt desktop frontend" OFF)
option(BEAMMP_GUI_ONLY "Build only the Qt frontend (for frontend development)" OFF)

if(NOT BEAMMP_GUI_ONLY)
    add_compile_definitions(CPPHTTPLIB_OPENSSL_SUPPORT)
    file(GLOB source_files CONFIGURE_DEPENDS "src/*.cpp" "src/*/*.cpp" "src/*/*.hpp" "include/*.h" "include/*/*.h" "include/*/*/*.h" "include/*.hpp" "include/*/*.hpp" "include/*/*/*.hpp")
    find_package(httplib CONFIG REQUIRED)
    find_package(nlohmann_json CONFIG REQUIRED)
    find_package(CURL REQUIRED)
    find_package(ZLIB REQUIRED)
    find_package(OpenSSL REQUIRED)
    add_executable(Launcher ${source_files})
    set_target_properties(Launcher PROPERTIES OUTPUT_NAME "BeamMP-Launcher")
    target_compile_definitions(Launcher PRIVATE $<$<CONFIG:Debug>:DEBUG>)
    target_link_libraries(Launcher PRIVATE ZLIB::ZLIB OpenSSL::SSL OpenSSL::Crypto
        httplib::httplib nlohmann_json::nlohmann_json CURL::libcurl)
    if(WIN32)
        target_link_libraries(Launcher PRIVATE ws2_32)
        set_property(TARGET Launcher PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    endif()
    target_include_directories(Launcher PRIVATE "include")
    target_compile_options(Launcher PRIVATE $<$<CXX_COMPILER_ID:MSVC>:/utf-8>)
    install(TARGETS Launcher RUNTIME DESTINATION .)
    if(BUILD_TESTING)
        find_package(Python3 REQUIRED COMPONENTS Interpreter)
        add_executable(backend-protocol-fixture tests/backend_fixture.cpp src/GuiBackend.cpp src/Options.cpp)
        target_include_directories(backend-protocol-fixture PRIVATE include)
        target_link_libraries(backend-protocol-fixture PRIVATE nlohmann_json::nlohmann_json CURL::libcurl OpenSSL::SSL OpenSSL::Crypto)
        if(WIN32)
            target_link_libraries(backend-protocol-fixture PRIVATE ws2_32)
            set_property(TARGET backend-protocol-fixture PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
        endif()
        add_test(NAME backend-protocol COMMAND Python3::Interpreter ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_backend_protocol.py $<TARGET_FILE:backend-protocol-fixture>)
    endif()
endif()

if(BEAMMP_BUILD_GUI OR BEAMMP_GUI_ONLY)
    add_subdirectory(gui)
endif()
